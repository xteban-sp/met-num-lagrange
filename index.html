def base_lagrange(xs, k, x):
    """L_k(x) = producto_{i != k} (x - x_i) / (x_k - x_i)"""
    Lk = 1.0
    for i in range(len(xs)):
        if i != k:
            Lk *= (x - xs[i]) / (xs[k] - xs[i])
    return Lk

def interpolar(xs, ys, x):
    """P_n(x) = suma_k y_k * L_k(x)   ->  O(n^2) con los dos bucles"""
    bases = [base_lagrange(xs, k, x) for k in range(len(xs))]
    return sum(ys[k] * bases[k] for k in range(len(xs))), bases

xs = [2, 4, 8, 12]          # memoria asignada (GB)
ys = [150, 85, 50, 70]      # latencia media (ms)
p, bases = interpolar(xs, ys, 6)
print("P3(6) =", p)        # 55.25 ms
